LN10 = math.log(10)

def ci(xs, a=0.05):
    n = len(xs); m = st.fmean(xs); s = st.stdev(xs); h = Z(1 - a / 2) * s / math.sqrt(n)
    return m, s, m - h, m + h, n


def trim(xs, p=0.1):
    xs = sorted(xs); k = int(len(xs) * p); return st.fmean(xs[k:len(xs) - k])


def f(x, d=4):
    return f"{x:+.{d}f}"

# ---------------- lambda ----------------
def lam_fit(get, hosts, ss, mode="cell"):
    """slope of y on log2(m); 'cell': within (host,seed) demeaning (host x seed FE)."""
    num = den = 0.0
    for h in hosts:
        xs = [math.log2(par[(h, a)] / par[(h, "no_growth")]) for a in ["no_growth"] + SC]
        xb = st.fmean(xs)
        for s in ss:
            ys = [get(h, a, s) for a in ["no_growth"] + SC]
            yb = st.fmean(ys)
            for x, y in zip(xs, ys):
                num += (x - xb) * (y - yb); den += (x - xb) ** 2
    return -num / den


gl = lambda h, a, s: late[(h, a)][s]

print("\n## J. Keep/cull with rollback, one decision (exploratory; dev set reused for decision and score)")

            dd.append((early, Lg - Ln))
        v1 = [k - a for k, a in zip(keepv, alw)]; v2 = [k - n_ for k, n_ in zip(keepv, nev)]
        m1, _, lo1, hi1, _ = ci(v1); m2, _, lo2, hi2, _ = ci(v2)
        cr = st.correlation([a for a, b in dd], [b for a, b in dd])
        sign = sum((a < 0) == (b < 0) for a, b in dd) / len(dd)
        print(h, f"d={d_ep}", f"rule-always_keep {m1:+.4f} [{lo1:+.4f},{hi1:+.4f}]", f"rule-never {m2:+.4f} [{lo2:+.4f},{hi2:+.4f}]",

              f"kept {sum(a<0 for a,b in dd)/len(dd):.2f}")

    g0 = [-x for x in d]; mu = st.fmean(g0); sd = st.stdev(g0)
    for n in (192, 768):

        normal = {g: 1 - NormalDist(g, se).cdf(0.05) for g in (0.04, 0.05, 0.06)}
        # nonparametric: shift observed per-seed gains to mean g, resample n
        boot = {}
        for g in (0.04, 0.06):
            sh = [x - mu + g for x in g0]; hits = 0
            for _ in range(2000):
                hits += st.fmean(random.choices(sh, k=n)) > 0.05
            boot[g] = hits / 2000
        lb_bind = 1.96 * sd / math.sqrt(n)
        print(lab, f"sd {sd:.3f} n={n} half-width {1.96*se:.4f} P(confirm|g): normal 0.04:{normal[0.04]:.3f} 0.05:{normal[0.05]:.3f} 0.06:{normal[0.06]:.3f};",
              f"shifted-bootstrap 0.04:{boot[0.04]:.3f} 0.06:{boot[0.06]:.3f}; LB>0 needs mean>{lb_bind:.4f}")
    print(lab, "n for 95% half-width 0.01:", math.ceil((1.96 * sd / 0.01) ** 2))
