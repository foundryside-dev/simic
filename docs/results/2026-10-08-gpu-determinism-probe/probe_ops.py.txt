"""Q4: which ops refuse deterministic mode on this build, for every host x seed."""

from __future__ import annotations

import json
import sys
import warnings

import torch

from experiments.kernel_demo import PATHOLOGIES, SEED_NAMES, Config, Slot, Stage, append_seed_group, augment, build_host, build_optimizer, build_seed, derive, enable_class1, make_generator, normalize_u8

enable_class1()  # first statement, before any CUDA context
device = sys.argv[1] if len(sys.argv) > 1 else "cuda:0"
results = {}
g = make_generator(1)
x_u8 = torch.randint(0, 256, (32, 3, 32, 32), generator=g, dtype=torch.uint8).to(device)
y = torch.randint(0, 10, (32,), generator=g).to(device)
crops = torch.randint(0, 9, (32, 2), generator=g, dtype=torch.uint8)
flips = torch.rand(32, generator=g) < 0.5

for pathology in PATHOLOGIES:
    for seed_name in (None, *SEED_NAMES):
        for stage in ([Stage.DORMANT] if seed_name is None else [Stage.TRAINING, Stage.BLENDING, Stage.FOSSILIZED]):
            key = f"{pathology}/{seed_name or 'no_seed'}/{stage.name}"
            try:
                host = build_host(pathology, derive(1, "h")).to(device)
                slot = Slot().to(device)
                opt = build_optimizer(host, Config())
                if seed_name:
                    slot.seed = build_seed(seed_name, 64, derive(1, "s")).to(device)
                    slot.seed.gain.data.fill_(0.5)
                    slot.stage = stage
                    slot.alpha = 0.5 if stage is Stage.BLENDING else 1.0
                    slot.beta = 1.0 if stage is Stage.FOSSILIZED else 0.0
                    append_seed_group(opt, slot.seed, Config())
                host.train()
                slot.train()
                with warnings.catch_warnings(record=True) as w:
                    warnings.simplefilter("always")
                    logits = host(augment(x_u8, crops, flips), slot)
                    loss = torch.nn.functional.cross_entropy(logits, y) + slot.trust_region_loss(Config())
                    loss.backward()
                    opt.step()
                    host.eval(); slot.eval()
                    with torch.no_grad():
                        host(normalize_u8(x_u8), slot)
                torch.cuda.synchronize()
                results[key] = {"ok": True, "warnings": sorted({str(m.message)[:160] for m in w})}
            except Exception as e:  # noqa: BLE001
                results[key] = {"ok": False, "error": f"{type(e).__name__}: {str(e)[:300]}"}

bad = {k: v for k, v in results.items() if not v["ok"] or v["warnings"]}
print(json.dumps({"device": torch.cuda.get_device_name(device), "torch": torch.__version__, "cudnn": torch.backends.cudnn.version(), "n_cases": len(results), "problems": bad}, indent=1))
