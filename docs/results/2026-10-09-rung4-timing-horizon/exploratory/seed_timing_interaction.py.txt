"""Exploratory, post-confirmatory: is there a seed-specific timing preference beyond residual noise?

Model per seed s over the five 10-epoch cells (graft epoch x in 0,1,2,3,5):
    y[s,x] = mu_s + tau_x + b_s * (x - xbar) + e[s,x]
tau_x is the shared timing curve; b_s is the seed's own linear timing slope around it.
If seeds share one timing response, var(b_s) is what the residual e alone implies: sigma^2 / Sxx.
Residual variance can include unstructured seed x timing scatter, which makes the ratio conservative.
Reads the per-run table given as argv[1]: .venv/bin/python <this file> tables/per-run.jsonl.gz
"""
import gzip, json, sys
import numpy as np
from scipy import stats

CELLS = {"T0": 0, "T1": 1, "T2": 2, "T3": 3, "T5": 5}
opener = gzip.open if sys.argv[1].endswith(".gz") else open
y = {}
for line in opener(sys.argv[1], "rt"):
    r = json.loads(line)
    if r["cell"] in CELLS and r["late_ce"]["scheduled"] is not None:
        y.setdefault(r["seed"], {})[CELLS[r["cell"]]] = r["late_ce"]["scheduled"]
xs = np.array(sorted(CELLS.values()), dtype=float)
Y = np.array([[y[s][x] for x in xs] for s in sorted(y) if len(y[s]) == len(xs)])
n, k = Y.shape
xc = xs - xs.mean()
sxx = float(xc @ xc)
R = Y - Y.mean(axis=1, keepdims=True)          # remove seed means
tau = R.mean(axis=0)                            # shared timing curve
R = R - tau
b = R @ xc / sxx                                # seed-specific slopes around the shared curve
E = R - np.outer(b, xc)
df_e = (n - 1) * (k - 2)
sigma2 = float((E**2).sum() / df_e)
pred_var = sigma2 / sxx
obs_var = float(b.var(ddof=1))
F = obs_var / pred_var
p = 1 - stats.f.cdf(F, n - 1, df_e)
best = np.argmin(Y, axis=1)
print(json.dumps({
    "n_seeds": n,
    "shared_curve_minus_mean": dict(zip([f"x{int(x)}" for x in xs], np.round(tau, 4).tolist())),
    "mean_slope_per_epoch": round(float((Y - Y.mean(axis=1, keepdims=True)) @ xc / sxx @ np.ones(n) / n), 4),
    "slope_sd_observed": round(obs_var**0.5, 4),
    "slope_sd_predicted_by_residual": round(pred_var**0.5, 4),
    "variance_ratio_F": round(F, 3), "df": [n - 1, df_e], "p_upper": round(float(p), 4),
    "residual_sd": round(sigma2**0.5, 4),
    "argmin_cell_share_winners_curse_caution": {f"x{int(xs[i])}": round(float(np.mean(best == i)), 3) for i in range(k)},
}, indent=1))

# --- Additions after the statistics review (still exploratory) ---
# Runs are deterministic, so the residual is the non-linear part of the seed x timing interaction,
# not replicate noise; the F reference distribution is not valid (heavy tails). Seed bootstrap instead.
rng = np.random.default_rng(20261009)


def fit(Ym):
    Rm = Ym - Ym.mean(axis=1, keepdims=True)
    Rm = Rm - Rm.mean(axis=0)
    bm = Rm @ xc / sxx
    Em = Rm - np.outer(bm, xc)
    s2 = (Em**2).sum() / ((Ym.shape[0] - 1) * (k - 2))
    return bm.var(ddof=1) / (s2 / sxx), max(bm.var(ddof=1) - s2 / sxx, 0.0) ** 0.5, Rm


Fs, sds = zip(*[fit(Y[rng.integers(0, n, n)])[:2] for _ in range(2000)])
_, _, Rint = fit(Y)
t0 = list(xs).index(0.0)
gain = Y[:, t0] - Y.min(axis=1)  # best-of-5 per seed against always-T0; inflated by the winner's curse
print(json.dumps({
    "bootstrap_F_95": [round(float(np.quantile(Fs, q)), 3) for q in (0.025, 0.975)],
    "bootstrap_implied_slope_sd_95": [round(float(np.quantile(sds, q)), 4) for q in (0.025, 0.975)],
    "interaction_sd_per_cell": round(float(Rint.std(ddof=1)), 4),
    "best_of_5_over_always_T0": {"mean": round(float(gain.mean()), 4), "median": round(float(np.median(gain)), 4)},
    "corr_T0mT1_vs_T2mT3": round(float(np.corrcoef(Y[:, 0] - Y[:, 1], Y[:, 2] - Y[:, 3])[0, 1]), 3),
    "corr_T0mT1_vs_T3mT5": round(float(np.corrcoef(Y[:, 0] - Y[:, 1], Y[:, 3] - Y[:, 4])[0, 1]), 3),
    "slope_kurtosis": round(float(stats.kurtosis(b)), 2),
}, indent=1))
