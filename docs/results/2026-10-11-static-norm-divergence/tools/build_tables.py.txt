def jsonl(path: Path) -> list[dict]:
    return [json.loads(line) for line in path.open()]


def arm_rows(rows: list[dict], arm: str) -> list[dict]:
    return [r for r in rows if r.get("arm") == arm]


def diverged(rows: list[dict], arm: str) -> dict | None:
    d = [r for r in arm_rows(rows, arm) if r["kind"] == "diverged"]
    return d[0] if d else None


def epochs(rows: list[dict], arm: str) -> list[dict]:
    return [r for r in arm_rows(rows, arm) if r["kind"] == "epoch"]


def snapshot_commit(study: str) -> str:
    launch = json.loads((RUNS / study / "launch.json").read_text())
    return launch["git"]["commit"][:7]


def unit_dirs(study: str) -> list[Path]:
    return sorted(p.parent for p in (RUNS / study).rglob("training.jsonl") if "src" not in p.relative_to(RUNS / study).parts[:1])


def section(title: str) -> None:
    print(f"\n## {title}\n")


def replay_checks() -> None:
    section("1. Replays reproduce the records bit for bit")
    print("| Replay | Study (snapshot commit) | Unit | Arm | Epoch hashes + dev equal | Divergence equal | Birth equal |")
    print("|---|---|---|---|---|---|---|")
    for tag, (study, unit, arm) in REPLAYS.items():
        rec = jsonl(RUNS / study / unit / "training.jsonl")
        rep = jsonl(PROBE / tag / "training.jsonl")
        ea = {r["epoch"]: r for r in epochs(rec, arm)}
        eb = {r["epoch"]: r for r in epochs(rep, arm)}
        same_epochs = sorted(ea) == sorted(eb) and all(
            ea[e]["training_state_sha256"] == eb[e]["training_state_sha256"] and ea[e]["dev"] == eb[e]["dev"] for e in ea
        )
        da, db = diverged(rec, arm), diverged(rep, arm)
        same_div = (da is None and db is None) or (
            da is not None and db is not None and (da["epoch"], da["step"], da["witness"]) == (db["epoch"], db["step"], db["witness"])
        )
        sa = [r for r in arm_rows(rec, arm) if r["kind"] == "arm_start"][0]
        sb = [r for r in arm_rows(rep, arm) if r["kind"] == "arm_start"][0]
        same_birth = sa["birth"] == sb["birth"] and sa["initial_dev"] == sb["initial_dev"]
        print(f"| {tag} | {study} ({snapshot_commit(study)}) | {unit} | {arm} | {same_epochs} ({len(ea)} epochs) | {same_div} | {same_birth} |")


def divergence_rates() -> None:
    section("2. Static-arm divergences, once per static trajectory")
    print("rung4-timing-horizon's T0-T5 directories share one 10-epoch static trajectory per seed (checked below);")
    print("H20 is a separate 20-epoch run. lifecycle-v2-validation's v1 and v2 share static (checked below).\n")
    counts: Counter = Counter()
    divs: Counter = Counter()
    shared_ok = True
    for study in STUDIES:
        for unit in unit_dirs(study):
            rel = unit.relative_to(RUNS / study).parts
            rows = jsonl(unit / "training.jsonl")
            if not arm_rows(rows, "static"):
                continue
            if study == "rung4-timing-horizon" and rel[-1] in ("T1", "T2", "T3", "T5"):
                t0 = jsonl(unit.parent / "T0" / "training.jsonl")
                if [(r["kind"], r.get("training_state_sha256"), r.get("step")) for r in arm_rows(rows, "static")] != [
                    (r["kind"], r.get("training_state_sha256"), r.get("step")) for r in arm_rows(t0, "static")
                ]:
                    shared_ok = False
                continue
            if study == "lifecycle-v2-validation" and rel[-1] == "v1":
                v2 = jsonl(unit.parent / "v2" / "training.jsonl")
                if [(r["kind"], r.get("training_state_sha256"), r.get("step")) for r in arm_rows(rows, "static")] != [
                    (r["kind"], r.get("training_state_sha256"), r.get("step")) for r in arm_rows(v2, "static")
                ]:
                    shared_ok = False
                continue
            manifest = json.loads((unit / "manifest.json").read_text())
            host = manifest["host"].removeprefix("kernel-demo-")  # top-level keys exist in every schema version
            key = (study, host, manifest["seed_type"], manifest["spec"]["epochs"])
            counts[key] += 1
            d = diverged(rows, "static")
            if d is not None:
                divs[key] += 1
    print(f"Shared static trajectories identical where claimed: {shared_ok}\n")
    print("| Study | Host | Seed | Epochs | Static diverged |")
    print("|---|---|---|---:|---:|")
    for key in sorted(counts):
        print(f"| {key[0]} | {key[1]} | {key[2]} | {key[3]} | {divs[key]}/{counts[key]} |")
    for epochs_ in (10, 20):
        n = sum(v for k, v in counts.items() if k[1] == "under_normalized" and k[2] == "norm" and k[3] == epochs_)
        k = sum(v for k_, v in divs.items() if k_[1] == "under_normalized" and k_[2] == "norm" and k_[3] == epochs_)
        print(f"\nunder_normalized x norm, {epochs_} epochs: {k}/{n}")
    n = sum(v for k, v in counts.items() if k[2] == "conv_heavy")
    k = sum(v for k_, v in divs.items() if k_[2] == "conv_heavy")
    print(f"conv_heavy static, all hosts and horizons: {k}/{n}")


def gain_curvature() -> None:

def basins() -> None:
    section("8. rung-4 H20 static arms by gain basin (exploratory, post hoc)")
    print("Basin: sign of the median over epochs 5-9 of each epoch's (gain_max + gain_min) / 2 witness.")
    print("Late rise: rung-4's definition (mean dev CE over the last three epochs > earlier minimum + 0.1).\n")
    runs = []
    for unit in unit_dirs("rung4-timing-horizon"):
        if unit.name != "H20":
            continue
        rows = jsonl(unit / "training.jsonl")
        ep = epochs(rows, "static")
        mids = [(e["witness"]["gain_max"] + e["witness"]["gain_min"]) / 2 for e in ep[5:10]]
        basin = "early" if not mids else ("+" if st.median(mids) > 0 else "-")
        d = diverged(rows, "static")
        devs = [e["dev"]["ce"] for e in ep]
        rise = d is None and len(devs) == 20 and st.mean(devs[-3:]) - min(devs[:-3]) > 0.1
        final_gain = (ep[-1]["witness"]["gain_max"] + ep[-1]["witness"]["gain_min"]) / 2 if ep else None
        runs.append({"basin": basin, "div": d, "rise": rise, "finite": d is None and len(devs) == 20, "final_gain": final_gain})
    finite = [r for r in runs if r["finite"]]
    print(f"Finite static runs: {len(finite)}; late risers: {sum(r['rise'] for r in finite)} (rung-4 report: 117/756)\n")
    print("| Basin | Runs | Diverged | Diverged at epoch >= 13 | Late rise (finite runs) |")
    print("|---|---:|---:|---:|---:|")
    for b in ("+", "-", "early"):
        R = [r for r in runs if r["basin"] == b]
        F = [r for r in R if r["finite"]]
        late_div = sum(1 for r in R if r["div"] and r["div"]["epoch"] >= 13)
        print(f"| {b} | {len(R)} | {sum(1 for r in R if r['div'])} | {late_div} | {sum(r['rise'] for r in F)}/{len(F)} |")
    a = sum(r["rise"] for r in finite if r["basin"] == "+")
    b_ = sum(1 for r in finite if r["basin"] == "+") - a
    c = sum(r["rise"] for r in finite if r["basin"] == "-")
    d_ = sum(1 for r in finite if r["basin"] == "-") - c
    n, r1, c1 = a + b_ + c + d_, a + b_, a + c
    p = sum(comb(r1, k) * comb(n - r1, c1 - k) / comb(n, c1) for k in range(a, min(r1, c1) + 1))
    print(f"\nFisher exact, one-sided (late rise more frequent in the + basin): p = {p:.2g}")
    plus = [r for r in finite if r["basin"] == "+"]
    print(
        f"+ basin final-epoch gain (median): risers {st.median(r['final_gain'] for r in plus if r['rise']):.3f}, "
        f"non-risers {st.median(r['final_gain'] for r in plus if not r['rise']):.3f}"
    )

def clopper_pearson(k: int, n: int, level: float = 0.95) -> tuple[float, float]:
    """Exact binomial interval by bisection: P(X >= k | lower) = a and P(X <= k | upper) = a."""
    a = (1 - level) / 2

    def at_most(p: float, upto: int) -> float:
        return sum(comb(n, i) * p**i * (1 - p) ** (n - i) for i in range(upto + 1))

    def bisect(f, target: float) -> float:  # f decreasing in p
        lo, hi = 0.0, 1.0
        for _ in range(80):
            mid = (lo + hi) / 2
            lo, hi = (mid, hi) if f(mid) > target else (lo, mid)
        return (lo + hi) / 2

    lower = 0.0 if k == 0 else bisect(lambda p: at_most(p, k - 1), 1 - a)
    upper = 1.0 if k == n else bisect(lambda p: at_most(p, k), a)
    return lower, upper


def fisher_one_sided(a: int, b: int, c: int, d: int) -> float:
    n, r1, c1 = a + b + c + d, a + b, a + c
    return sum(comb(r1, k) * comb(n - r1, c1 - k) / comb(n, c1) for k in range(a, min(r1, c1) + 1))

def late_onset() -> None:
    section("11. Late-onset cap against a sham, from the same onset (exploratory)")
    print("Onset: the first step, within the final 60 probed steps before the recorded divergence, at which the gain's step exceeds 0.1.")
    print("Cap: the gain's step capped at 0.1 from the onset (momentum buffer clipped to match). Sham: the gain's step scaled by 0.999 from the onset.")
    print("Prefix: probed steps up to and including the onset step whose gain, CE, h_ms and dCE/dgain equal the uncapped replay's exactly")
    print("(the intervention acts at the onset step's optimiser update, after that step is probed).\n")
    print("| Seed | Onset | Probed steps from onset to recorded divergence | Recorded | Sham | Cap | Sham final dev CE | Cap final dev CE | No-growth final dev CE (record) | Prefix steps identical (sham, cap) |")
    print("|---|---|---:|---|---|---|---:|---:|---:|---|")
    discordant = {"cap_only": 0, "sham_only": 0}
    for line in (HERE / "tools" / "replays-late-cap.txt").read_text().splitlines():
        study, unit, arm, _power, tag, onset = line.split()
        rec = jsonl(RUNS / study / unit / "training.jsonl")
        e0 = int(onset.split(":")[0])
        out = []
        prefix = []
        for kind in ("late-sham-", "late-cap-"):
            t = tag.replace("late-cap-", kind)
            path = PROBE / t / "training.jsonl"
            if not path.exists():
                out.append(("missing", None))
                prefix.append("-")
                continue
            rows = jsonl(path)
            d = diverged(rows, arm)
            ep = epochs(rows, arm)
            out.append((f"diverged e{d['epoch']} s{d['step']}" if d else f"completed ({len(ep)} ep)", None if d else ep[-1]["dev"]["ce"]))
            base_rows = jsonl(PROBE / {"7074": "gc2n-7074-static-v2"}.get(tag.split("-")[-1], tag.removeprefix("late-cap-")) / "probe.jsonl")
            mine = jsonl(PROBE / t / "probe.jsonl")
            e_on, s_on = (int(v) for v in onset.split(":"))
            k_on = next(i for i, r in enumerate(base_rows) if (r["epoch"], r["step"]) == (e_on, s_on))
            keys = ("epoch", "step", "gain", "ce", "h_ms", "dce_dg")
            same = sum(all(a[x] == b[x] for x in keys) for a, b in zip(base_rows[: k_on + 1], mine[: k_on + 1], strict=False))
            prefix.append(f"{same}/{k_on + 1}")
        d = diverged(rec, arm)
        cap_dev = "-" if out[1][1] is None else f"{out[1][1]:.3f}"
        sham_dev = "-" if out[0][1] is None else f"{out[0][1]:.3f}"
        ng = epochs(rec, "no_growth")[-1]["dev"]["ce"]
        sham_ok, cap_ok = out[0][0].startswith("completed"), out[1][0].startswith("completed")
        discordant["cap_only"] += cap_ok and not sham_ok
        discordant["sham_only"] += sham_ok and not cap_ok
        base = jsonl(PROBE / {"7074": "gc2n-7074-static-v2"}.get(tag.split("-")[-1], tag.removeprefix("late-cap-")) / "probe.jsonl")
        e_on, s_on = (int(v) for v in onset.split(":"))
        k_on = next(i for i, r in enumerate(base) if (r["epoch"], r["step"]) == (e_on, s_on))
        print(f"| {tag.removeprefix('late-cap-')} | e{onset.replace(':', ' s')} | {len(base) - k_on} | diverged e{d['epoch']} s{d['step']} | {out[0][0]} | {out[1][0]} | {sham_dev} | {cap_dev} | {ng:.3f} | {prefix[0]}, {prefix[1]} |")
    def mcnemar(b: int, c: int) -> float:
        return min(1.0, 2 * sum(comb(b + c, k) for k in range(max(b, c), b + c + 1)) / 2 ** (b + c)) if b + c else 1.0

    b, c = discordant["cap_only"], discordant["sham_only"]
    print(f"\nDiscordant pairs: cap survives and sham fails {b}; sham survives and cap fails {c}. Exact McNemar (two-sided) p = {mcnemar(b, c):.3g}.")
    print(f"Counting 8346's capped run (final dev CE at chance) as not rescued: {b - 1} against {c}, p = {mcnemar(b - 1, c):.3g}.")
